cmake_minimum_required(VERSION 3.20)
project(v_mixer_offline VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(reac_core src/protocol.cpp)
target_include_directories(reac_core PUBLIC include)
target_compile_features(reac_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(reac_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(reac_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(reac-inspect src/inspect.cpp)
target_link_libraries(reac-inspect PRIVATE reac_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(reac-tests tests/protocol_tests.cpp)
  target_link_libraries(reac-tests PRIVATE reac_core)
  foreach(case codec ordering ambiguity envelope control robustness)
    add_test(NAME core_${case} COMMAND reac-tests ${case})
  endforeach()
  find_package(Python3 3.9 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli_import COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py $<TARGET_FILE:reac-inspect>)
    set(REAC_PUBLIC_FIXTURES "" CACHE FILEPATH "External pinned real_frames.json (never copied into source tree)")
    if(REAC_PUBLIC_FIXTURES)
      add_test(NAME public_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_public_reference.py $<TARGET_FILE:reac-inspect> ${REAC_PUBLIC_FIXTURES})
    endif()
  endif()
endif()
